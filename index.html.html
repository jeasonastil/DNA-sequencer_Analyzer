function loadFASTA() { const file = document.getElementById("fastaFile").files[0]; if (!file) { alert("Please select a FASTA file!"); return; } const reader = new FileReader(); reader.onload = function(event) { let text = event.target.result; let sequence = text .split("\n") .filter(line => !line.startsWith(">")) .join("") .replace(/\s/g, "") .toUpperCase(); document.getElementById("dnaSequence").value = sequence; alert("FASTA sequence loaded successfully!"); }; reader.readAsText(file); }
